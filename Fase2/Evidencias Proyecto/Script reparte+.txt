-- =========================
-- 1) TABLAS BASE
-- =========================

CREATE TABLE participante (
    participante_id     SERIAL PRIMARY KEY,
    email               VARCHAR(255),
    nombre              VARCHAR(255) NOT NULL,
    telefono            VARCHAR(50),
    es_usuario          BOOLEAN DEFAULT FALSE,
    fecha_creacion      TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW()
);


CREATE TABLE grupo (
    grupo_id        SERIAL PRIMARY KEY,
    nombre          VARCHAR(255) NOT NULL,
    descripcion     TEXT,
    fecha_inicio    DATE,
    fecha_cierre    DATE,
    fecha_creacion  TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),
    creado_por      INTEGER NOT NULL,
    estado          VARCHAR(50) DEFAULT 'ABIERTO',

    CONSTRAINT fk_grupo_creado_por
        FOREIGN KEY (creado_por)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


CREATE TABLE usuario (
    usuario_id      SERIAL PRIMARY KEY,
    participante_id INTEGER NOT NULL,
    sub_cognito     VARCHAR(255) NOT NULL UNIQUE,
    fecha_registro  TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),

    CONSTRAINT fk_usuario_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE expo_tokens (
    id              SERIAL PRIMARY KEY,
    user_id         INTEGER NOT NULL,
    expo_token      VARCHAR(500) NOT NULL,
    updated_at      TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),

    CONSTRAINT fk_expotokens_usuario
        FOREIGN KEY (user_id)
        REFERENCES usuario (usuario_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE tasa_cambio_moneda (
    moneda                  VARCHAR(10) PRIMARY KEY,   -- p.ej. 'CLP', 'USD'
    moneda_base             VARCHAR(10) NOT NULL,      -- p.ej. 'CLP'
    tasa_base_mayor         NUMERIC(18,6) NOT NULL,
    fecha_actualizacion     TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    proveedor               VARCHAR(100)
);


-- =========================
-- 2) GASTOS / BOLETAS / PAGOS
-- =========================

CREATE TABLE gastos (
    gasto_id                SERIAL PRIMARY KEY,
    grupo_id                INTEGER NOT NULL,
    participante_id         INTEGER NOT NULL,      -- quién registra
    participantegasto_id    INTEGER,              -- si lo usas para algo especial
    descripciongasto        VARCHAR(500),
    moneda                  VARCHAR(10) NOT NULL, -- FK a tasa_cambio_moneda
    monto                   NUMERIC(18,2) NOT NULL,
    fecha_registro          TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),
    estado                  VARCHAR(50) DEFAULT 'ACTIVO',
    decimales               INTEGER DEFAULT 0,
    tasa_cambio_usada       NUMERIC(18,6),
    fuente_tasa_cambio      VARCHAR(100),
    fecha_tasa_cambio       TIMESTAMP WITHOUT TIME ZONE,
    monto_clp_al_cierre     NUMERIC(18,2),
    convertido_a_clp        BOOLEAN DEFAULT FALSE,

    CONSTRAINT fk_gasto_grupo
        FOREIGN KEY (grupo_id)
        REFERENCES grupo (grupo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_gasto_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_gasto_moneda
        FOREIGN KEY (moneda)
        REFERENCES tasa_cambio_moneda (moneda)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


CREATE TABLE boletas (
    id              SERIAL PRIMARY KEY,
    s3_bucket       VARCHAR(255),
    s3_key          VARCHAR(500),
    vendor          VARCHAR(255),
    fecha           DATE,
    total           NUMERIC(18,2),
    currency        VARCHAR(10),
    items           JSONB,
    parsed_json     JSONB,
    raw_text        TEXT,
    gasto_id        INTEGER,

    CONSTRAINT fk_boleta_gasto
        FOREIGN KEY (gasto_id)
        REFERENCES gastos (gasto_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE pago (
    id              SERIAL PRIMARY KEY,
    gasto_id        INTEGER NOT NULL,
    participante_id INTEGER NOT NULL,     -- quién paga
    monto_pagado    NUMERIC(18,2) NOT NULL,
    fecha           TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),

    CONSTRAINT fk_pago_gasto
        FOREIGN KEY (gasto_id)
        REFERENCES gastos (gasto_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_pago_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
);


-- =========================
-- 3) RELACIONES GRUPO / PARTICIPANTES
-- =========================

CREATE TABLE grupo_participante (
    grupo_id        INTEGER NOT NULL,
    participante_id INTEGER NOT NULL,

    PRIMARY KEY (grupo_id, participante_id),

    CONSTRAINT fk_gp_grupo
        FOREIGN KEY (grupo_id)
        REFERENCES grupo (grupo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_gp_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE invitaciones (
    invitacion_id       SERIAL PRIMARY KEY,
    participante_id     INTEGER,           -- destinatario (puede ser null si aún no existe)
    token               VARCHAR(255) NOT NULL UNIQUE,
    estado              VARCHAR(50) DEFAULT 'PENDIENTE',
    email               VARCHAR(255),
    nombre              VARCHAR(255),
    telefono            VARCHAR(50),
    grupo_id            INTEGER NOT NULL,
    grupo_nombre        VARCHAR(255),
    fecha_invitacion    TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),
    fecha_respuesta     TIMESTAMP WITHOUT TIME ZONE,
    tipo                VARCHAR(50),
    expires_at          TIMESTAMP WITHOUT TIME ZONE,

    CONSTRAINT fk_invitacion_grupo
        FOREIGN KEY (grupo_id)
        REFERENCES grupo (grupo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_invitacion_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE SET NULL
);


-- =========================
-- 4) ASIGNACIÓN DE GASTOS A PERSONAS
-- =========================

CREATE TABLE gasto_participante (
    id                  SERIAL PRIMARY KEY,
    gasto_id            INTEGER NOT NULL,
    participante_id     INTEGER NOT NULL,
    monto_asignado      NUMERIC(18,2) NOT NULL,
    created_at          TIMESTAMP WITHOUT TIME ZONE DEFAULT NOW(),

    CONSTRAINT fk_gastop_gasto
        FOREIGN KEY (gasto_id)
        REFERENCES gastos (gasto_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_gastop_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


-- =========================
-- 5) LIQUIDACIONES
-- =========================

CREATE TABLE liquidacion_grupo (
    grupo_id            INTEGER PRIMARY KEY,
    moneda_base         VARCHAR(10) NOT NULL,
    fecha_liquidacion   TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    total_base_minima   NUMERIC(18,2) NOT NULL,
    detalle_resumen     TEXT,

    CONSTRAINT fk_liqgrupo_grupo
        FOREIGN KEY (grupo_id)
        REFERENCES grupo (grupo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);


CREATE TABLE liquidacion_grupo_participante (
    grupo_id                INTEGER NOT NULL,
    participante_id         INTEGER NOT NULL,
    asignado_base_minima    NUMERIC(18,2) NOT NULL,
    pagado_base_minima      NUMERIC(18,2) NOT NULL,
    saldo_base_minima       NUMERIC(18,2) NOT NULL,

    PRIMARY KEY (grupo_id, participante_id),

    CONSTRAINT fk_liqgp_grupo
        FOREIGN KEY (grupo_id)
        REFERENCES grupo (grupo_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    CONSTRAINT fk_liqgp_participante
        FOREIGN KEY (participante_id)
        REFERENCES participante (participante_id)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);
